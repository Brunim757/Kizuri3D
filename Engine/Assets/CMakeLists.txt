if(WIN32)
  file(GLOB KZ_SKY_HDR CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/Samples/Assets/Skies/*.hdr)
  set(KZ_SKY_RC ${KZ_FBS_GEN_DIR}/EmbeddedSkies.rc)
  set(KZ_SKY_CPP ${KZ_FBS_GEN_DIR}/EmbeddedSkiesWin.cpp)
  file(WRITE ${KZ_SKY_RC} "// auto-generated: embedded HDR skies\n")
  file(WRITE ${KZ_SKY_CPP} "// auto-generated: embedded HDR sky names\n#include \"Kizuri/Assets/EmbeddedSkies.h\"\n#define NOMINMAX\n#define WIN32_LEAN_AND_MEAN\n#include <windows.h>\nnamespace Kizuri {\n")
  set(KZ_SKY_ID 101)
  set(KZ_SKY_NAMES "")
  foreach(KZ_SKY_F IN LISTS KZ_SKY_HDR)
    get_filename_component(KZ_SKY_NM "${KZ_SKY_F}" NAME)
    file(TO_NATIVE_PATH "${KZ_SKY_F}" KZ_SKY_NAT)
    file(APPEND ${KZ_SKY_RC} "${KZ_SKY_ID} RCDATA \"${KZ_SKY_NAT}\"\n")
    set(KZ_SKY_NAMES "${KZ_SKY_NAMES}  {\"${KZ_SKY_NM}\", ${KZ_SKY_ID}},\n")
    math(EXPR KZ_SKY_ID "${KZ_SKY_ID} + 1")
  endforeach()
  file(APPEND ${KZ_SKY_CPP} "static const struct SkyNameId { const char* name; int id; } kSkies[] = {\n${KZ_SKY_NAMES}};\nsize_t EmbeddedSkyCount() { return sizeof(kSkies) / sizeof(kSkies[0]); }\nconst EmbeddedSkyEntry& EmbeddedSkyAt(size_t i) {\n  static EmbeddedSkyEntry e = { \"\", nullptr, 0 };\n  size_t n = EmbeddedSkyCount();\n  size_t k = i < n ? i : 0;\n  e.name = n > 0 ? kSkies[k].name : \"\";\n  e.data = nullptr;\n  e.size = 0;\n  if (n > 0) {\n    HMODULE mod = GetModuleHandle(nullptr);\n    if (mod != nullptr) {\n      HRSRC r = FindResource(mod, MAKEINTRESOURCE(kSkies[k].id), RT_RCDATA);\n      if (r != nullptr) {\n        HGLOBAL g = LoadResource(mod, r);\n        if (g != nullptr) {\n          e.size = static_cast<size_t>(SizeofResource(mod, r));\n          e.data = static_cast<const unsigned char*>(LockResource(g));\n        }\n      }\n    }\n  }\n  return e;\n}\n} // namespace Kizuri\n")
  set(KZ_SKY_SRCS ${KZ_SKY_RC} ${KZ_SKY_CPP})
else()
  set(KZ_SKY_SRCS src/EmbeddedSkiesNull.cpp)
endif()
